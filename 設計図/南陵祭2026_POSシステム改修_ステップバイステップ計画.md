### `functions/index.js` の変更詳細

#### 【削除】`getNextReceiptNumber` (エクスポート関数)

**現状**: クライアントから直接呼び出し可能な `onCall` 関数として公開されている。

```javascript
exports.getNextReceiptNumber = functions.region(...).https.onCall(...)
```

**変更後**: クライアントへの公開をやめ、`functions/index.js` 内部でのみ使うプライベートな `async function` に変更する。

```javascript
// プライベートユーティリティ関数（エクスポートなし）
async function getNextReceiptNumber(channel, transaction) {
  const ranges = {
    pos:    { min: 100,  max: 999  },
    mobile: { min: 7000, max: 7999 },
    sok:    { min: 2000, max: 2999 },
  };
  const { min, max } = ranges[channel];
  const rangeSize = max - min + 1;

  const counterRef = db.doc(`counters/receipt_${channel}`);
  const counterSnap = await transaction.get(counterRef);
  let candidate = (counterSnap.data()?.current ?? min - 1) + 1;

  // アクティブな注文と衝突しない番号を探す
  for (let attempt = 0; attempt < rangeSize; attempt++) {
    if (candidate > max) candidate = min;

    // アクティブ判定: completed / cancelled / abandoned 以外
    const activeStatuses = ["cooking", "ready_to_serve", "ready_for_pickup"];
    const dupQuery = db.collection("orders")
      .where("receiptNumber", "==", candidate)
      .where("status", "in", activeStatuses);
    const dupSnap = await transaction.get(dupQuery);

    if (dupSnap.empty) {
      transaction.set(counterRef, { current: candidate, updatedAt: admin.firestore.FieldValue.serverTimestamp() }, { merge: true });
      return candidate;
    }
    candidate++;
  }
  throw new functions.https.HttpsError("resource-exhausted", "受付番号の空きがありません");
}
```

**変更理由**:
- 設計憲法§4.1: `getNextReceiptNumber` は独立した OnCall 関数ではなく、`createOrder` などの内部で呼ぶユーティリティ。
- クライアントに直接呼ばせることはセキュリティ上好ましくない。
- 旧版は `counters/receipt` の単一ドキュメントに `currentNumber`, `currentNumber_MOBILE`, `currentNumber_SOK` フィールドを持っていたが、新版は経路ごとに独立したドキュメント（`counters/receipt_pos` 等）を使う。

#### 【新規追加】`createOrder`

```javascript
exports.createOrder = functions.region("asia-northeast1").https.onCall(async (data, context) => {
  if (!context.auth) throw new functions.https.HttpsError("unauthenticated", "ログインが必要です。");

  const { orderChannel, storeId, items } = data;

  // --- バリデーション ---
  if (!["mobile", "pos"].includes(orderChannel))
    throw new functions.https.HttpsError("invalid-argument", "orderChannel は mobile または pos である必要があります。");
  if (!storeId || !Array.isArray(items) || items.length === 0)
    throw new functions.https.HttpsError("invalid-argument", "storeId と items が必要です。");

  const uid = context.auth.uid;

  // mobile はドメイン制限
  if (orderChannel === "mobile") {
    const email = context.auth.token.email || "";
    if (!email.endsWith("@gl.pen-kanagawa.ed.jp") && email !== "ynrcs1000@gmail.com")
      throw new functions.https.HttpsError("permission-denied", "モバイルオーダーは在校生のみ利用可能です。");

    // BAN チェック
    const banDoc = await db.doc(`banned_users/${uid}`).get();
    if (banDoc.exists) throw new functions.https.HttpsError("permission-denied", "利用が制限されています。");
  }

  // pos は store_admin 権限チェック
  if (orderChannel === "pos") {
    const token = context.auth.token;
    if (token.role !== "store_admin" || token.storeId !== storeId)
      throw new functions.https.HttpsError("permission-denied", "店舗管理者権限が必要です。");
  }

  // --- 商品情報取得 & 金額サーバーサイド計算 ---
  const itemRefs = items.map(i => db.doc(`items/${i.itemId}`));
  const productDocs = await db.getAll(...itemRefs);
  const productMap = new Map(productDocs.filter(d => d.exists).map(d => [d.id, d.data()]));

  let totalPrice = 0;
  const orderItems = [];
  for (const item of items) {
    const product = productMap.get(item.itemId);
    if (!product) continue;
    if (product.storeId !== storeId)
      throw new functions.https.HttpsError("permission-denied", "他店舗の商品は注文できません。");
    if (!product.isAvailable)
      throw new functions.https.HttpsError("failed-precondition", `「${product.name}」は売り切れです。`);
    totalPrice += product.price * item.quantity;
    orderItems.push({
      itemId: item.itemId,
      name: product.name,
      price: product.price,
      quantity: item.quantity,
      customizations: item.customizations || [],
    });
  }

  // --- トランザクション: 受付番号発番 + 注文作成 ---
  const result = await db.runTransaction(async (tx) => {
    const receiptNumber = await getNextReceiptNumber(orderChannel, tx);

    const orderRef = db.collection("orders").doc();
    const now = admin.firestore.FieldValue.serverTimestamp();
    const orderData = {
      status: "cooking",                     // ← 旧: "authorized" / "unpaid_at_pos"
      orderChannel,                          // ← 旧: フィールドなし / "pos"
      storeId,
      items: orderItems,
      totalPrice,
      receiptNumber,
      userId:    orderChannel === "mobile" ? uid : null,    // ← 旧: POSでも createdBy に入れていた
      createdBy: orderChannel === "pos"    ? uid : null,
      paymentMethod: "au_pay_manual",        // ← 旧: "ONLINE" / "pos"
      sokStatus: null,
      cancellationReason: null,
      note: null,
      termsAgreedAt: orderChannel === "mobile" ? (data.termsAgreedAt || now) : null,
      createdAt: now,
      updatedAt: now,
      readyToServeAt: null,
      readyForPickupAt: null,
      completedAt: null,
      cancelledAt: null,
      abandonedAt: null,
    };
    tx.set(orderRef, orderData);
    return { orderId: orderRef.id, receiptNumber };
  });

  return { success: true, orderId: result.orderId, receiptNumber: result.receiptNumber };
});
```

**主な変更点まとめ**:

| 項目 | 旧 | 新 |
|---|---|---|
| 関数名 | `createOnlineOrder` / `createPOSOrder` | `createOrder` (統合) |
| `status` 初期値 | `"authorized"` / `"unpaid_at_pos"` | `"cooking"` |
| `orderChannel` | なし / `"pos"` | `"mobile"` / `"pos"` |
| `paymentMethod` | `"ONLINE"` / `"pos"` | `"au_pay_manual"` (統一) |
| `userId` | mobile: UID, POS: なし | mobile: UID, POS: `null` |
| `createdBy` | なし | pos: UID, mobile: `null` |
| カート読み込み | `users/{uid}/cart` から読む | クライアントから `items[]` を受け取る |
| BAN チェック | なし | `banned_users/{uid}` を確認 |
| 受付番号カウンター | `counters/receipt` の単一ドキュメント | `counters/receipt_{channel}` の経路別ドキュメント |

### `firestore.rules` の変更詳細

**現状**: ルールの内容は不明だが、クライアントから `orders` に `update()` が通っていた（presenter.html が直接 `status: "ready_for_pickup"` に更新していた）。

**変更後**:

```javascript
match /orders/{orderId} {
  // 読み取り
  allow read: if
    // 自身の注文
    (request.auth != null && request.auth.uid == resource.data.userId)
    // SOK 未確定仮注文 (claim前)
    || (resource.data.orderChannel == "sok" && resource.data.sokStatus == "pending")
    // monitor.html 用 (ready_for_pickup のみ)
    || resource.data.status == "ready_for_pickup";

  // 作成・更新・削除はすべてクライアントから禁止
  allow create, update, delete: if false;
}

match /counters/{docId} {
  allow read: if request.auth != null && request.auth.token.role == "store_admin";
  allow write: if false; // Functions 経由のみ
}
```

**変更理由**: 設計憲法§8の要件に準拠。特に `orders` の write は Functions 経由のみとすることで、クライアントによる不正なステータス改ざんを防ぐ。

  - 本番環境にデプロイし、クライアントから `createOrder` を呼び出して正しく発番され `cooking` で保存されることを確認。
